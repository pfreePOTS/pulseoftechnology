<!-- ═══ INTAKE SURVEY SCRIPT ═══════════════════════════════════════════ -->
<script>
(function(){
  var answers = {};
  var progressBar = document.getElementById('intake-progress');
  var allStepIds = ['step-0','step-1','step-2','step-3','step-4','step-result'];

  function setProgress(pct){ if(progressBar) progressBar.style.width = pct + '%'; }

  function showStep(id){
    allStepIds.forEach(function(s){
      var el = document.getElementById(s);
      if(el) el.classList.remove('active');
    });
    var t = document.getElementById(id);
    if(t) t.classList.add('active');
  }

  function advanceTo(nextId, progressPct){
    setTimeout(function(){ setProgress(progressPct); showStep(nextId); }, 280);
  }

  // ── MAP region click ──────────────────────────────────────────────────────
  document.querySelectorAll('.map-region').forEach(function(g){
    g.addEventListener('click', function(){
      document.querySelectorAll('.map-region').forEach(function(r){ r.classList.remove('selected'); });
      this.classList.add('selected');
      answers.location = this.dataset.region;
      advanceTo('step-1', 40);
    });
  });

  // ── Map text input ────────────────────────────────────────────────────────
  var mapSubmit = document.getElementById('map-state-submit');
  if(mapSubmit){
    mapSubmit.addEventListener('click', function(){
      var val = (document.getElementById('map-state-input').value || '').trim();
      if(val){ answers.location = val; advanceTo('step-1', 40); }
    });
  }
  var mapInput = document.getElementById('map-state-input');
  if(mapInput){
    mapInput.addEventListener('keydown', function(e){
      if(e.key === 'Enter'){ var val = this.value.trim(); if(val){ answers.location = val; advanceTo('step-1', 40); } }
    });
  }

  // ── Multiple-choice options ───────────────────────────────────────────────
  var stepFlow = {
    industry: { next: 'step-2', progress: 60 },
    role:     { next: 'step-3', progress: 80 },
    issue:    { next: 'step-4', progress: 80 }
  };

  document.querySelectorAll('.intake-option').forEach(function(btn){
    btn.addEventListener('click', function(){
      var key = this.dataset.key;
      var val = this.dataset.value;

      // Highlight selection
      var parent = this.closest('.intake-options');
      if(parent) parent.querySelectorAll('.intake-option').forEach(function(b){ b.classList.remove('selected'); });
      this.classList.add('selected');

      // Show "Other" text input if needed
      if(val === 'Other' || val === 'Something Else'){
        var otherWrap = document.getElementById('other-' + key);
        if(otherWrap){
          otherWrap.style.display = 'flex';
          var inp = otherWrap.querySelector('.intake-other-input');
          if(inp) inp.focus();
        }
        return; // wait for text submit
      }

      answers[key] = val;
      var flow = stepFlow[key];
      if(flow) advanceTo(flow.next, flow.progress);
    });
  });

  // ── "Other" text submit buttons ───────────────────────────────────────────
  document.querySelectorAll('.intake-other-submit').forEach(function(btn){
    btn.addEventListener('click', function(){
      var key  = this.dataset.key;
      var nextId = this.dataset.next;
      var prog = parseInt(this.dataset.progress);
      var inp  = document.getElementById('other-' + key + '-input');
      var val  = inp ? inp.value.trim() : '';
      if(!val) return;
      answers[key] = val;
      advanceTo(nextId, prog);
    });
  });

  // ── Open-ended submit ─────────────────────────────────────────────────────
  function showResult(){
    setProgress(100);
    var ctaMap = {
      'Cybersecurity': 'our Cybersecurity Assessment',
      'AI':            'our AI Readiness Assessment',
      'Compliance':    'our Compliance Readiness Assessment',
      'Cloud':         'our Cloud & Infrastructure solutions',
      'IT Management': 'our Managed IT Services',
      'Strategy':      'our Strategic Advisory services'
    };
    var cta = ctaMap[answers.issue] || 'our recommended solutions';
    var sub = 'Based on what you shared, we\'d suggest starting with ' + cta + '.';
    if(answers.location) sub += ' Our team serving the ' + answers.location + ' area is ready to help.';
    var msgEl = document.getElementById('intake-result-msg');
    var subEl = document.getElementById('intake-result-sub');
    if(msgEl) msgEl.textContent = 'Thanks \u2014 we know just where to start.';
    if(subEl) subEl.textContent = sub;
    showStep('step-result');
  }

  var sitSubmit = document.getElementById('situation-submit');
  if(sitSubmit){
    sitSubmit.addEventListener('click', function(){
      answers.situation = (document.getElementById('situation-input').value || '').trim();
      showResult();
    });
  }
  var sitSkip = document.getElementById('situation-skip');
  if(sitSkip){ sitSkip.addEventListener('click', function(){ showResult(); }); }

  // ── Restart ───────────────────────────────────────────────────────────────
  var restartBtn = document.getElementById('intake-restart');
  if(restartBtn){
    restartBtn.addEventListener('click', function(){
      answers = {};
      setProgress(20);
      document.querySelectorAll('.intake-option').forEach(function(b){ b.classList.remove('selected'); });
      document.querySelectorAll('.intake-other-wrap').forEach(function(w){ w.style.display = 'none'; });
      document.querySelectorAll('.intake-other-input').forEach(function(i){ i.value = ''; });
      document.querySelectorAll('.map-region').forEach(function(r){ r.classList.remove('selected'); });
      var mapIn = document.getElementById('map-state-input'); if(mapIn) mapIn.value = '';
      var sitIn = document.getElementById('situation-input'); if(sitIn) sitIn.value = '';
      showStep('step-0');
    });
  }
})();
</script>
